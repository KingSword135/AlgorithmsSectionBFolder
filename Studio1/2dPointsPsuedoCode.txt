function getDistance(point1, point2):
    return sqrt((point1[0] - point2[0])^2 + (point2[0] - point2[1])^2)

function getClosestPairs(points_array):

    size = LENGTH_OF(points_array)
    mindist = 9999
    minpoints = NULL

    FOR i = 0 TO size:
        FOR j = i + 1 TO size:
            IF getDistance(points_array[i],points_array[j] < mindist):
                mindist = getDistance(points_array[i],points_array[j])
                minpoints = (points_array[i],points_array[j])

    RETURN mindist AND minpoints
